      return true;
    }

      return false;

    const userAgent = window.navigator.userAgent.toLowerCase();

    }

  const getStoredTheme = () => {

      const storedTheme = localStorage.getItem("marketing-theme-v2");
      return storedTheme === "light" || storedTheme === "dark" ? storedTheme : null;
    } catch {
      return null;
    }
  };

  const resolveSystemTheme = () => (systemDarkMediaQuery.matches ? "dark" : "light");

  const applyTheme = (resolvedTheme) => {
    classList.remove("light", "dark");

  };

  const syncThemeWithCurrentPreference = () => {
    if (isBionicRoute()) {

      return;

    applyTheme(storedTheme ?? resolveSystemTheme());
  };

  syncDocsTheme();
  syncThemeWithCurrentPreference();

<ol>
<li>Download and run local LLMs like gpt-oss or Llama, Qwen</li>

<h2 class="group scroll-m-28" id="system-requirements"><span>System requirements</span><a aria-label="Link to section system-requirements" data-card="" href="#system-requirements" class="text-fd-muted-foreground ml-2 inline-flex align-middle opacity-0 transition-opacity group-hover:opacity-100"><svg xmlns="http://www.w3.org/2000/svg" width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" class="lucide lucide-link size-3.5" aria-hidden="true"><path d="M10 13a5 5 0 0 0 7.54.54l3-3a5 5 0 0 0-7.07-7.07l-1.72 1.71"></path><path d="M14 11a5 5 0 0 0-7.54-.54l-3 3a5 5 0 0 0 7.07 7.07l1.71-1.71"></path></svg></a></h2>
<p>LM Studio generally supports Apple Silicon Macs, x64/ARM64 Windows PCs, and x64 Linux PCs.</p>
<p>Consult the <a href="app/system-requirements">System Requirements</a> page for more detailed information.</p>
<h2 class="group scroll-m-28" id="run-llamacpp-gguf-or-mlx-models"><span>Run llama.cpp (GGUF) or MLX models</span><a aria-label="Link to section run-llamacpp-gguf-or-mlx-models" data-card="" href="#run-llamacpp-gguf-or-mlx-models" class="text-fd-muted-foreground ml-2 inline-flex align-middle opacity-0 transition-opacity group-hover:opacity-100"><svg xmlns="http://www.w3.org/2000/svg" width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" class="lucide lucide-link size-3.5" aria-hidden="true"><path d="M10 13a5 5 0 0 0 7.54.54l3-3a5 5 0 0 0-7.07-7.07l-1.72 1.71"></path><path d="M14 11a5 5 0 0 0-7.54-.54l-3 3a5 5 0 0 0 7.07 7.07l1.71-1.71"></path></svg></a></h2>
<p>LM Studio supports running LLMs on Mac, Windows, and Linux using <a href="https://github.com/ggerganov/llama.cpp" rel="noreferrer noopener" target="_blank"><code>llama.cpp</code></a>.</p>
<p>On Apple Silicon Macs, LM Studio also supports running LLMs using Apple&#x27;s <a href="https://github.com/ml-explore/mlx" rel="noreferrer noopener" target="_blank"><code>MLX</code></a>.</p>
<p>To install or manage LM Runtimes, press <code>⌘</code> <code>Shift</code> <code>R</code> on Mac or <code>Ctrl</code> <code>Shift</code> <code>R</code> on Windows/Linux.</p>

<p>You can install MCP servers in LM Studio and use them with your local models.</p>
<p>See the docs for more: <a href="/docs/app/plugins/mcp">Use MCP server</a>.</p>
<p>If you&#x27;re develping an MCP server, check out <a href="/docs/app/plugins/mcp/deeplink">Add to LM Studio Button</a>.</p>
<h2 class="group scroll-m-28" id="run-an-llm-like-gpt-oss-llama-qwen-mistral-or-deepseek-r1-on-your-computer"><span>Run an LLM like <code>gpt-oss</code>, <code>Llama</code>, <code>Qwen</code>, <code>Mistral</code>, or <code>DeepSeek R1</code> on your computer</span><a aria-label="Link to section run-an-llm-like-gpt-oss-llama-qwen-mistral-or-deepseek-r1-on-your-computer" data-card="" href="#run-an-llm-like-gpt-oss-llama-qwen-mistral-or-deepseek-r1-on-your-computer" class="text-fd-muted-foreground ml-2 inline-flex align-middle opacity-0 transition-opacity group-hover:opacity-100"><svg xmlns="http://www.w3.org/2000/svg" width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" class="lucide lucide-link size-3.5" aria-hidden="true"><path d="M10 13a5 5 0 0 0 7.54.54l3-3a5 5 0 0 0-7.07-7.07l-1.72 1.71"></path><path d="M14 11a5 5 0 0 0-7.54-.54l-3 3a5 5 0 0 0 7.07 7.07l1.71-1.71"></path></svg></a></h2>

<p>You can do this right within LM Studio! See <a href="app/basics/download-model">Download an LLM</a> for guidance.</p>
